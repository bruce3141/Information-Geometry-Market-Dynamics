% Entropy-style header-includes for pandoc -> XeLaTeX -> PDF
% Approximates the MDPI/Entropy journal print style without claiming to be the
% official template. Targets:
%   - Palatino body text (mathpazo) for classic journal feel
%   - A4 paper with MDPI-like margins (narrow)
%   - Section heads in a saturated accent color
%   - booktabs tables, caption package for journal-style captions
%   - hyperref with colored links in the accent

% ---------- Math and symbol packages ----------
\usepackage{amsmath,amssymb,amsthm}
\usepackage{mathtools}  % handles \tag{} reliably

% ---------- Fonts ----------
% Palatino is installed via pandoc's `fontfamily=mathpazo` variable (see
% build_pdf.py), which suppresses pandoc's default lmodern injection.
\linespread{1.05}

% ---------- Microtypography ----------
\usepackage{microtype}

% ---------- Graphics and color ----------
\usepackage{graphicx}
\usepackage[dvipsnames,table]{xcolor}

% Entropy-ish accent: deep navy / teal-blue as used in MDPI section heads
\definecolor{EntropyAccent}{HTML}{004A8B}
\definecolor{EntropyRule}{HTML}{004A8B}
\definecolor{EntropyLink}{HTML}{004A8B}

% ---------- Page geometry (MDPI-like narrow margins) ----------
\usepackage[a4paper,top=1.8cm,bottom=2.2cm,left=2.0cm,right=2.0cm,footskip=1.0cm]{geometry}

% ---------- Tables ----------
\usepackage{booktabs}
\usepackage{longtable}
\usepackage{array}

% ---------- Captions ----------
\usepackage[labelfont={bf,color=EntropyAccent},font=small,labelsep=period,justification=raggedright,singlelinecheck=false]{caption}

% ---------- Section heading style ----------
\usepackage{titlesec}
\titleformat{\section}
  {\color{EntropyAccent}\normalfont\large\bfseries}
  {\thesection.}{0.6em}{}
\titleformat{\subsection}
  {\color{EntropyAccent}\normalfont\normalsize\bfseries}
  {\thesubsection.}{0.6em}{}
\titleformat{\subsubsection}
  {\color{EntropyAccent}\normalfont\normalsize\itshape}
  {\thesubsubsection.}{0.6em}{}

\titlespacing*{\section}{0pt}{1.6ex plus 0.8ex minus 0.4ex}{0.6ex plus 0.3ex}
\titlespacing*{\subsection}{0pt}{1.2ex plus 0.6ex minus 0.3ex}{0.4ex plus 0.2ex}

% ---------- Running head and footer ----------
\usepackage{fancyhdr}
\pagestyle{fancy}
\fancyhf{}
\fancyhead[L]{\small\color{EntropyAccent}\itshape Information Geometry of Market Dynamics}
\fancyhead[R]{\small\color{EntropyAccent} \thepage\ of \pageref{LastPage}}
\fancyfoot[C]{\small\color{EntropyAccent} Dean, B.H. (2026), Draft v0.57}
\renewcommand{\headrulewidth}{0.4pt}
\renewcommand{\footrulewidth}{0pt}

% Need lastpage for "x of N" in header
\usepackage{lastpage}

% ---------- Hyperlinks ----------
\usepackage{hyperref}
\hypersetup{
  colorlinks=true,
  linkcolor=EntropyLink,
  citecolor=EntropyLink,
  urlcolor=EntropyLink,
  pdftitle={Information Geometry of Market Dynamics: A Pareto Frontier from Contact Geometry},
  pdfauthor={Bruce H. Dean}
}

% ---------- Abstract framing (boxed, left-accent bar) ----------
% A lightweight abstract frame: colored vertical rule on the left, smaller font.
\usepackage{mdframed}
\newmdenv[
  leftmargin=0pt,
  rightmargin=0pt,
  innerleftmargin=10pt,
  innerrightmargin=10pt,
  innertopmargin=8pt,
  innerbottommargin=8pt,
  topline=false,
  rightline=false,
  bottomline=false,
  linecolor=EntropyAccent,
  linewidth=3pt,
  backgroundcolor=white,
  skipabove=8pt,
  skipbelow=10pt
]{entropyabstract}

% Title block: reduce vertical waste. Pandoc renders \maketitle; customise below.
\usepackage{titling}
\pretitle{\begin{flushleft}\color{EntropyAccent}\Large\bfseries}
\posttitle{\par\end{flushleft}\vspace{-0.4em}}
\preauthor{\begin{flushleft}\normalsize}
\postauthor{\par\end{flushleft}\vspace{-0.5em}}
\predate{\begin{flushleft}\small\color{gray}}
\postdate{\par\end{flushleft}}

% Figure float placement: prefer here, then top; discourage bottom-only
\makeatletter
\renewcommand{\fps@figure}{htbp}
\renewcommand{\fps@table}{htbp}
\makeatother

% Tight spacing around display math
\setlength{\abovedisplayskip}{6pt}
\setlength{\belowdisplayskip}{6pt}
\setlength{\abovedisplayshortskip}{4pt}
\setlength{\belowdisplayshortskip}{4pt}

% Paragraph spacing like the journal: no indent, small vertical skip.
\setlength{\parindent}{0pt}
\setlength{\parskip}{0.5ex plus 0.3ex minus 0.2ex}

% Figure width control: cap at text width
\setkeys{Gin}{width=\linewidth,keepaspectratio}
